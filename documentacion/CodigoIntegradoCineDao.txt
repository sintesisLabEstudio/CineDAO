            using System;
            namespace CineStarBD.Dominio;
            // Entidad de dominio: modela una venta de boleteria con propiedades encapsuladas,
            // sobrecarga de constructores y metodos puros de calculo (sin efectos secundarios).
                CLASE: VentaCine
            public class VentaCine
            {
                public int Id { get; set; }                       // PK identity de dbo.Ventas.
                public string Cliente { get; set; } = "";         // Nombre del cliente.
                public string Rut { get; set; } = "";             // RUT del cliente.
                public bool EsSocio { get; set; }                 // Socio CineClub: 15% de descuento.
                public string PeliculaTexto { get; set; } = "";   // Pelicula elegida en cartelera.
                public int PrecioPelicula { get; set; }           // Precio unitario de la entrada.
                public int CantidadTickets { get; set; }          // Cantidad de entradas.
                public string ComboNombre { get; set; } = "Sin combo"; // Combo de confiteria opcional.
                public int PrecioCombo { get; set; }              // Precio del combo.
                public int SubtotalEntradas { get; set; }         // PrecioPelicula * CantidadTickets.
                public int SubtotalSnacks { get; set; }           // Precio del combo.
                public int MontoDescuento { get; set; }           // 15% del subtotal si es socio.
                public int NetoAfecto { get; set; }               // Subtotal - descuento.
                public int MontoIva { get; set; }                 // 19% del neto.
                public int TotalPagar { get; set; }               // Neto + IVA.
                // Constructor por defecto (necesario para el data binding y los DAO).
                public VentaCine() { }
                // Constructor sobrecargado con los datos de entrada de la boleteria.
                public VentaCine(string cliente, string rut, bool esSocio, string pelicula,
                                 int precioPelicula, int cantidadTickets, string comboNombre, int precioCombo)
                {
                    Cliente = cliente;
                    Rut = rut;
                    EsSocio = esSocio;
                    PeliculaTexto = pelicula;
                    PrecioPelicula = precioPelicula;
                    CantidadTickets = cantidadTickets;
                    ComboNombre = comboNombre;
                    PrecioCombo = precioCombo;
                    CalcularTotales();
                }
                // Metodos puros: solo leen propiedades y devuelven un valor.
                    METODO/FUNCION: public int CalcularSubtotalEntradas() => PrecioPelicula * CantidadTickets;
                public int CalcularSubtotalEntradas() => PrecioPelicula * CantidadTickets;
                    METODO/FUNCION: public int CalcularSubtotalSnacks() => PrecioCombo;
                public int CalcularSubtotalSnacks() => PrecioCombo;
                    METODO/FUNCION: public int CalcularMontoDescuento()
                public int CalcularMontoDescuento()
                    => EsSocio ? (int)((CalcularSubtotalEntradas() + CalcularSubtotalSnacks()) * 0.15) : 0;
                    METODO/FUNCION: public int CalcularNetoAfecto()
                public int CalcularNetoAfecto()
                    => CalcularSubtotalEntradas() + CalcularSubtotalSnacks() - CalcularMontoDescuento();
                    METODO/FUNCION: public int CalcularMontoIVA() => (int)(CalcularNetoAfecto() * 0.19);
                public int CalcularMontoIVA() => (int)(CalcularNetoAfecto() * 0.19);
                    METODO/FUNCION: public int CalcularTotal() => CalcularNetoAfecto() + CalcularMontoIVA();
                public int CalcularTotal() => CalcularNetoAfecto() + CalcularMontoIVA();
                // Materializa los valores calculados en las propiedades de la entidad.
                    METODO/FUNCION: public void CalcularTotales()
                public void CalcularTotales()
                {
                    SubtotalEntradas = CalcularSubtotalEntradas();
                    SubtotalSnacks = CalcularSubtotalSnacks();
                    MontoDescuento = CalcularMontoDescuento();
                    NetoAfecto = CalcularNetoAfecto();
                    MontoIva = CalcularMontoIVA();
                    TotalPagar = CalcularTotal();
                }
            }

            using System.Collections.Generic;
            using System.Linq;
            namespace CineStarBD.Dominio;
            // Reglas de negocio y catalogos de la boleteria (sin dependencia de WinForms).
            // Equivale a CalculadoraPedido en el proyecto Pizzeria.
                CLASE: CatalogoCine
            public static class CatalogoCine
            {
                private const string PeliculaPorDefecto = "Aventura 2D";
                private const string ComboPorDefecto = "Sin combo";
                // Cartelera: pelicula -> precio unitario de la entrada (CLP).
                public static readonly IReadOnlyDictionary<string, int> Peliculas = new Dictionary<string, int>
                {
                    { "Aventura 2D", 5000 },
                    { "Ciencia Ficcion 3D", 7000 },
                    { "Drama Premium", 6000 },
                    { "Estreno VIP", 9000 }
                };
                // Confiteria: combo -> recargo (CLP).
                public static readonly IReadOnlyDictionary<string, int> Combos = new Dictionary<string, int>
                {
                    { "Sin combo", 0 },
                    { "Combo Clasico", 3000 },
                    { "Combo Grande", 4500 },
                    { "Combo Familiar", 6000 }
                };
                // Datos de entrada tomados de la interfaz.
                CLASE: DatosVenta
                public sealed class DatosVenta
                {
                    public string Cliente { get; set; } = "";
                    public string Rut { get; set; } = "";
                    public bool EsSocio { get; set; }
                    public string Pelicula { get; set; } = PeliculaPorDefecto;
                    public string Combo { get; set; } = ComboPorDefecto;
                    public int CantidadTickets { get; set; } = 1;
                }
                // Aplica las reglas y devuelve una VentaCine con todos los montos calculados.
                    METODO/FUNCION: public static VentaCine Calcular(DatosVenta d)
                public static VentaCine Calcular(DatosVenta d)
                {
                    string pelicula = Peliculas.ContainsKey(d.Pelicula) ? d.Pelicula : PeliculaPorDefecto;
                    string combo = Combos.ContainsKey(d.Combo) ? d.Combo : ComboPorDefecto;
                    int cantidad = d.CantidadTickets > 0 ? d.CantidadTickets : 1;
                    return new VentaCine(d.Cliente, d.Rut, d.EsSocio, pelicula,
                                         Peliculas[pelicula], cantidad, combo, Combos[combo]);
                }
                // Utilidad para la grilla: nombres de peliculas ordenados.
                    METODO/FUNCION: public static string[] ObtenerPeliculas() => Peliculas.Keys.OrderBy(x => x).ToArray();
                public static string[] ObtenerPeliculas() => Peliculas.Keys.OrderBy(x => x).ToArray();
                // Utilidad para la grilla: nombres de combos ordenados.
                    METODO/FUNCION: public static string[] ObtenerCombos() => Combos.Keys.OrderBy(x => x).ToArray();
                public static string[] ObtenerCombos() => Combos.Keys.OrderBy(x => x).ToArray();
            }

            using System;

            public static class ConexionBD
            {
                // SQL Server Express segun la documentacion tecnica del caso CineStar.
                private static readonly string cadenaConexion =
                    @"Server=.\SQLEXPRESS;Database=CineStarDB;Integrated Security=True;TrustServerCertificate=True;";
                public static string CadenaConexion => cadenaConexion;
                // Abre y devuelve una conexion nueva. Quien la recibe debe liberarla con "using".
                    METODO/FUNCION: public static SqlConnection ObtenerConexion()
                public static SqlConnection ObtenerConexion()
                {
                    SqlConnection conexion = new SqlConnection(cadenaConexion);
                    try
                    {
                        conexion.Open();
                        return conexion;
                    }
                    catch
                    {
                        conexion.Dispose();
                        throw;
                    }
                }
                // Verifica que la base CineStarDB abra correctamente.
                    METODO/FUNCION: public static bool ProbarConexion()
                public static bool ProbarConexion()
                {
                    try
                    {
                        using (SqlConnection conexion = ObtenerConexion())
                            return conexion.State == ConnectionState.Open;
                    }
                    catch (SqlException ex)
                    {
                        Console.WriteLine($"[Error de Conexion SQL]: {ex.Message}");
                        return false;
                    }
                }
            }

            using System;
            using System.Data;
            using Microsoft.Data.SqlClient;
            namespace CineStarBD.Dominio;
            // DAO de dbo.Ventas: CRUD con consultas parametrizadas y transacciones.
                CLASE: VentaDAO
            public class VentaDAO
            {
                // Inserta una venta. El desglose se graba junto con la venta en una sola transaccion.
                    METODO/FUNCION: public bool InsertarVenta(VentaCine v)
                public bool InsertarVenta(VentaCine v)
                {
                    const string query = @"
                        INSERT INTO dbo.Ventas
                            (Cliente, Rut, EsSocio, PeliculaTexto, PrecioPelicula, CantidadTickets,
                             ComboNombre, PrecioCombo, SubtotalEntradas, SubtotalSnacks,
                             MontoDescuento, NetoAfecto, MontoIva, TotalPagar)
                        VALUES
                            (@Cliente, @Rut, @EsSocio, @PeliculaTexto, @PrecioPelicula, @CantidadTickets,
                             @ComboNombre, @PrecioCombo, @SubtotalEntradas, @SubtotalSnacks,
                             @MontoDescuento, @NetoAfecto, @MontoIva, @TotalPagar);
                        SELECT CAST(SCOPE_IDENTITY() AS INT);";
                    using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                        // -> FX: ObtenerConexion | ConexionBD.cs | Abre y devuelve una SqlConnection nueva.
                    using (SqlTransaction tran = conexion.BeginTransaction())
                    {
                        try
                        {
                            using (SqlCommand cmd = new SqlCommand(query, conexion, tran))
                            {
                                CargarParametros(cmd, v);
                                object? resultado = cmd.ExecuteScalar();
                                bool exito = resultado != null && resultado != DBNull.Value;
                                if (exito)
                                {
                                    v.Id = Convert.ToInt32(resultado);
                                    tran.Commit();
                                }
                                else
                                {
                                    Revertir(tran);
                                }
                                return exito;
                            }
                        }
                        catch
                        {
                            Revertir(tran);
                            throw;
                        }
                    }
                }
                // Devuelve el historial completo en un DataTable desconectado para el DataGridView.
                    METODO/FUNCION: public DataTable ObtenerVentasDesconectado()
                public DataTable ObtenerVentasDesconectado()
                {
                    DataTable dt = new DataTable();
                    const string query = @"SELECT Id, Cliente, Rut, EsSocio, PeliculaTexto,
                                                  CantidadTickets, ComboNombre, TotalPagar
                                           FROM dbo.Ventas ORDER BY Id DESC;";
                    using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                    using (SqlDataAdapter adapter = new SqlDataAdapter(query, conexion))
                    {
                        adapter.Fill(dt);
                    }
                    return dt;
                }
                // Filtra por cliente usando LIKE (clausula parametrizada).
                    METODO/FUNCION: public DataTable BuscarVentasPorCliente(string filtroCliente)
                public DataTable BuscarVentasPorCliente(string filtroCliente)
                {
                    DataTable dt = new DataTable();
                    const string query = @"SELECT Id, Cliente, Rut, EsSocio, PeliculaTexto,
                                                  CantidadTickets, ComboNombre, TotalPagar
                                           FROM dbo.Ventas WHERE Cliente LIKE @Filtro ORDER BY Id DESC;";
                    using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                    using (SqlCommand cmd = new SqlCommand(query, conexion))
                    {
                        cmd.Parameters.AddWithValue("@Filtro", "%" + filtroCliente + "%");
                        using (SqlDataAdapter adapter = new SqlDataAdapter(cmd))
                        {
                            adapter.Fill(dt);
                        }
                    }
                    return dt;
                }
                // Obtiene una venta completa por su Id, o null si no existe.
                    METODO/FUNCION: public VentaCine? ObtenerVentaPorId(int id)
                public VentaCine? ObtenerVentaPorId(int id)
                {
                    const string query = @"SELECT Id, Cliente, Rut, EsSocio, PeliculaTexto, PrecioPelicula,
                                                  CantidadTickets, ComboNombre, PrecioCombo, SubtotalEntradas,
                                                  SubtotalSnacks, MontoDescuento, NetoAfecto, MontoIva, TotalPagar
                                           FROM dbo.Ventas WHERE Id = @Id;";
                    using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                    using (SqlCommand cmd = new SqlCommand(query, conexion))
                    {
                        cmd.Parameters.AddWithValue("@Id", id);
                        using (SqlDataReader r = cmd.ExecuteReader())
                        {
                            return r.Read() ? LeerVenta(r) : null;
                        }
                    }
                }
                // Actualiza una venta existente dentro de una transaccion.
                    METODO/FUNCION: public bool ActualizarVenta(VentaCine v)
                public bool ActualizarVenta(VentaCine v)
                {
                    const string query = @"UPDATE dbo.Ventas SET
                                             Cliente=@Cliente, Rut=@Rut, EsSocio=@EsSocio,
                                             PeliculaTexto=@PeliculaTexto, PrecioPelicula=@PrecioPelicula,
                                             CantidadTickets=@CantidadTickets, ComboNombre=@ComboNombre,
                                             PrecioCombo=@PrecioCombo, SubtotalEntradas=@SubtotalEntradas,
                                             SubtotalSnacks=@SubtotalSnacks, MontoDescuento=@MontoDescuento,
                                             NetoAfecto=@NetoAfecto, MontoIva=@MontoIva, TotalPagar=@TotalPagar
                                           WHERE Id=@Id;";
                    using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                    using (SqlTransaction tran = conexion.BeginTransaction())
                    {
                        try
                        {
                            using (SqlCommand cmd = new SqlCommand(query, conexion, tran))
                            {
                                cmd.Parameters.AddWithValue("@Id", v.Id);
                                CargarParametros(cmd, v);
                                bool exito = cmd.ExecuteNonQuery() > 0;
                                if (exito) tran.Commit();
                                else Revertir(tran);
                                return exito;
                            }
                        }
                        catch
                        {
                            Revertir(tran);
                            throw;
                        }
                    }
                }
                // Borrado fisico previa confirmacion en la interfaz.
                    METODO/FUNCION: public bool EliminarVenta(int id)
                public bool EliminarVenta(int id)
                {
                    const string query = "DELETE FROM dbo.Ventas WHERE Id=@Id;";
                    using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                    using (SqlCommand cmd = new SqlCommand(query, conexion))
                    {
                        cmd.Parameters.AddWithValue("@Id", id);
                        return cmd.ExecuteNonQuery() > 0;
                    }
                }
                // Metricas globales acumuladas en tiempo real.
                    METODO/FUNCION: public DataTable ObtenerMetricas()
                public DataTable ObtenerMetricas()
                {
                    DataTable dt = new DataTable();
                    const string query = @"SELECT COUNT(*) AS TotalVentas,
                                                  ISNULL(SUM(TotalPagar), 0) AS TotalRecaudado,
                                                  ISNULL(AVG(TotalPagar), 0) AS PromedioVenta
                                           FROM dbo.Ventas;";
                    using (SqlConnection conexion = ConexionBD.ObtenerConexion())
                    using (SqlDataAdapter adapter = new SqlDataAdapter(query, conexion))
                    {
                        adapter.Fill(dt);
                    }
                    return dt;
                }

                    METODO/FUNCION: private static VentaCine LeerVenta(SqlDataReader r) => new VentaCine
                private static VentaCine LeerVenta(SqlDataReader r) => new VentaCine

                    Cliente = r["Cliente"]?.ToString() ?? "",
                    Rut = r["Rut"]?.ToString() ?? "",
                    EsSocio = Convert.ToBoolean(r["EsSocio"]),
                    PeliculaTexto = r["PeliculaTexto"]?.ToString() ?? "",
                    PrecioPelicula = Convert.ToInt32(r["PrecioPelicula"]),
                    CantidadTickets = Convert.ToInt32(r["CantidadTickets"]),
                    ComboNombre = r["ComboNombre"]?.ToString() ?? "Sin combo",
                    PrecioCombo = Convert.ToInt32(r["PrecioCombo"]),
                    SubtotalEntradas = Convert.ToInt32(r["SubtotalEntradas"]),
                    SubtotalSnacks = Convert.ToInt32(r["SubtotalSnacks"]),
                    MontoDescuento = Convert.ToInt32(r["MontoDescuento"]),
                    NetoAfecto = Convert.ToInt32(r["NetoAfecto"]),
                    MontoIva = Convert.ToInt32(r["MontoIva"]),
                    TotalPagar = Convert.ToInt32(r["TotalPagar"])
                };
                    METODO/FUNCION: private static void CargarParametros(SqlCommand cmd, VentaCine v)
